cmake_minimum_required(VERSION 3.20)
project(radaer_native VERSION 0.5.12 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(Threads REQUIRED)
add_library(radaer_core src/core.cpp src/protocol.cpp src/monitor.cpp src/report.cpp src/shared_packets.cpp src/ipv4_reassembly.cpp src/discovery_range.cpp
  src/ue.cpp src/ue_movement.cpp src/ue_properties.cpp src/health.cpp src/equipment.cpp src/loot.cpp src/players.cpp src/battle.cpp src/rates.cpp src/radar_replay.cpp)
target_include_directories(radaer_core PUBLIC include third_party)
target_link_libraries(radaer_core PUBLIC OpenSSL::Crypto Threads::Threads)
if(MSVC)
  target_compile_options(radaer_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(radaer_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(radaer-native src/main.cpp src/capture.cpp src/shared_windows.cpp src/lan_discovery.cpp src/linux_network.cpp src/capture_firewall.cpp src/webui.cpp)
target_compile_definitions(radaer-native PRIVATE RADAER_WEB_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/webui" RADAER_REPO_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/..")
target_link_libraries(radaer-native PRIVATE radaer_core)
if(WIN32)
  target_compile_definitions(radaer-native PRIVATE FD_SETSIZE=1024)
  target_link_libraries(radaer-native PRIVATE ws2_32 iphlpapi advapi32 ole32 oleaut32 uuid shell32)
endif()
include(CTest)
if(BUILD_TESTING)
add_executable(radaer-health-tests tests/health_tests.cpp)
target_link_libraries(radaer-health-tests PRIVATE radaer_core)
add_test(NAME health_state_lifecycle COMMAND radaer-health-tests)
if(WIN32)
  add_library(radaer-windivert-fixture SHARED tests/mock_windivert.cpp)
  target_include_directories(radaer-windivert-fixture PRIVATE third_party)
endif()
add_executable(radaer-tests tests/core_tests.cpp)
target_link_libraries(radaer-tests PRIVATE radaer_core)
add_test(NAME native_core COMMAND radaer-tests)
add_executable(radaer-bitstream-tests tests/bitstream_tests.cpp)
target_link_libraries(radaer-bitstream-tests PRIVATE radaer_core)
add_test(NAME bitstream_boundaries COMMAND radaer-bitstream-tests)
add_executable(radaer-battle-tests tests/battle_tests.cpp)
target_link_libraries(radaer-battle-tests PRIVATE radaer_core)
add_test(NAME battle_session_lifecycle COMMAND radaer-battle-tests)
add_executable(radaer-rate-tests tests/rate_tests.cpp)
target_link_libraries(radaer-rate-tests PRIVATE radaer_core)
add_test(NAME live_rate_metrics COMMAND radaer-rate-tests)
add_executable(radaer-shared-tests tests/shared_tests.cpp)
target_link_libraries(radaer-shared-tests PRIVATE radaer_core)
add_test(NAME shared_packet_routing COMMAND radaer-shared-tests)
add_executable(radaer-ipv4-reassembly-tests tests/ipv4_reassembly_tests.cpp)
target_link_libraries(radaer-ipv4-reassembly-tests PRIVATE radaer_core)
add_test(NAME ipv4_udp_reassembly COMMAND radaer-ipv4-reassembly-tests)
add_executable(radaer-player-probe tests/player_probe.cpp)
target_link_libraries(radaer-player-probe PRIVATE radaer_core)
add_executable(radaer-player-benchmark EXCLUDE_FROM_ALL tools/player_benchmark.cpp)
target_link_libraries(radaer-player-benchmark PRIVATE radaer_core)
foreach(target radaer-native radaer-tests)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
